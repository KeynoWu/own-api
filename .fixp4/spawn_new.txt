            std::thread::spawn(move || {
                while let Ok(ev) = rx.blocking_recv() {
                    match ev {
                        tauri_plugin_shell::process::CommandEvent::Stderr(line) => {
                            eprintln!("[sidecar] {}", String::from_utf8_lossy(&line));
                        }
                        tauri_plugin_shell::process::CommandEvent::Error(e) => {
                            eprintln!("[sidecar] {e}");
                        }
                        _ => {}
                    }
                }
