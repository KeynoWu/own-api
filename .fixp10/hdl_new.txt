      if (v === 'overview' && ovSt.refresh > 0) timer = setInterval(() => go('overview'), ovSt.refresh * 1000);
    });
}

addEventListener('unhandledrejection', (e) => { toast('请求失败：' + ((e.reason && e.reason.message) || String(e.reason)), true); e.preventDefault(); });
