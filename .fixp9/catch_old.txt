        try {
          renameSync(DB_FILE, backup);
        } catch {
          /* ignore */
        }
        console.error(`[store] db.json 解析失败，已备份到 ${backup}，使用空库启动`, err);
      } else {
        console.error('[store] db.json 装载异常（不判损坏、不改名，空库兜底启动，原文件保留）：', err);
      }
